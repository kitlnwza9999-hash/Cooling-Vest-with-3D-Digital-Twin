🔌 ตารางกำหนดขาพอร์ต ESP32 (GPIO Pinout)
GPIO สัญญาณ ประเภท อุปกรณ์ปลายทาง
GPIO 18 FAN_PWM PWM Out (25kHz) MOSFET ขับพัดลมเอวคู่ 12V (70mm)
GPIO 19 FAN_TACH Digital In (Interrupt) สัญญาณความเร็วรอบพัดลมจริง (RPM)
GPIO 26 SHOULDER_PWM PWM Out (25kHz) MOSFET ขับพัดลมบ่าไหล่เป่าหน้าคู่ (40mm)
GPIO 25 TEC_PWM PWM Out (40kHz) วงจรขับ Peltier TEC1-12706 (Spine)
GPIO 39 BATT_NTC Analog In (ADC1_CH3) NTC 10K วัดอุณหภูมิเซลล์แบตเตอรี่ (BTMS)
GPIO 21 I2C_SDA I2C Bus (Data) BME280, MAX30102, MPU6050
GPIO 22 I2C_SCL I2C Bus (Clock) BME280, MAX30102, MPU6050
GPIO 34 BATT_SENSE Analog In (ADC) วงจรแบ่งแรงดัน (Voltage Divider 12V)
GPIO 4 BUZZER Digital Out บัซเซอร์เตือนภัยฉุกเฉิน 85dB
GPIO 2 HAPTIC Digital Out มอเตอร์สั่นเตือนที่คอเสื้อ (Vibration SOS)
🛡️ มาตรการความปลอดภัยทางไฟฟ้าและสรีรวิทยา
  • การป้องกันความร้อนสะสมในแบตเตอรี่ (BTMS): ติดตั้งแผ่นกราฟีนเกลี่ยความร้อน ฮีตซิงก์อลูมิเนียม พัดลมดูดระบายความร้อนโดยเฉพาะ และ NTC Sensor บน GPIO 39 ตัดไฟหรือเร่งระบายความร้อนหากอุณหภูมิเกิน 45°C
  • การป้องกันเหงื่อและความชื้น (IP54/IP65): เคลือบน้ำยา Silicone Conformal Coating บนแผงวงจรทั้งหมด และใช้กล่องบรรจุภัณฑ์แบบกันน้ำพร้อมยางซีลซิลิโคน
  • ป้องกันการลัดวงจร (Short Circuit Protection): ติดตั้ง Resettable PTC Fuse ขนาด 5A ตัดวงจรภายในเวลาต่ำกว่า 50ms เมื่อเกิดการลัดวงจร
  • การจัดระเบียบสายไฟ (Harness Routing): ใช้สายไฟแกนทองแดงชุบดีบุกหุ้มซิลิโคนทนความร้อนสูง (Silicone Wire 18/20 AWG สำหรับไฟหลัก, 26 AWG สำหรับสัญญาณ) เดินร้อยในท่อผ้าป้องกันการเสียดสี

💻 เฟิร์มแวร์ ESP32 สำหรับควบคุมระบบปรับอากาศ (Production C++ Code)

โค้ดสถาปัตยกรรม Dual-Core FreeRTOS ควบคุมพัดลม 4 ตัว 2 โซน, ระบบหล่อเย็นแบตเตอรี่ BTMS, คำนวณ Heat Index, ป้องกันจุดน้ำค้าง (Dew Point Interlock) และเชื่อมต่อ Digital Twin

/**
 * ==============================================================================
 * Project: Smart Cooling Vest with 3D Digital Twin Telemetry & Dual-Zone Turbofans
 * Microcontroller: ESP32-WROOM-32E (240MHz Dual-Core)
 * Framework: Arduino IDE / PlatformIO with FreeRTOS
 * Author: Antigravity Engineering Team
 * ==============================================================================
 */

#include <Arduino.h>
#include <Wire.h>
#include <Adafruit_Sensor.h>
#include <Adafruit_BME280.h>
#include <BLEDevice.h>
#include <BLEServer.h>
#include <BLEUtils.h>
#include <BLE2902.h>

// ================= PIN DEFINITIONS =================
#define PIN_FAN_PWM          18   // PWM พัดลมหอยโข่งคู่ลำตัวและหลัง (70mm 12V)
#define PIN_FAN_TACH         19   // สัญญาณวัดรอบพัดลม (Tachometer Interrupt)
#define PIN_SHOULDER_PWM     26   // PWM พัดลมไมโครเทอร์โบติดบ่าไหล่เป่าหน้า/คอ (40mm 12V)
#define PIN_TEC_PWM          25   // PWM ควบคุมแผ่นเย็น Peltier TEC1-12706
#define PIN_I2C_SDA          21   // I2C Data สำหรับเซนเซอร์สภาพแวดล้อม
#define PIN_I2C_SCL          22   // I2C Clock
#define PIN_BATT_SENSE       34   // ADC วัดระดับแรงดันแบตเตอรี่ (Voltage Divider)
#define PIN_BATT_NTC         39   // ADC1_CH3 วัดอุณหภูมิผิวเซลล์แบตเตอรี่ (BTMS)
#define PIN_HEATSINK_NTC     35   // ADC วัดอุณหภูมิฮีตซิงก์แผ่น Peltier
#define PIN_BUZZER            4   // บัซเซอร์เตือนภัยฉุกเฉิน
#define PIN_HAPTIC            2   // มอเตอร์สั่นเตือนที่คอเสื้อ

// ================= PWM CHANNELS & CONSTANTS =================
#define PWM_FREQ_FAN         25000 // 25kHz High Frequency (เงียบ ไร้เสียงฮัม)
#define PWM_FREQ_TEC         40000 // 40kHz สำหรับวงจรกรอง LC Filter
#define PWM_RES              8     // 8-bit Resolution (0 - 255)
#define PWM_CH_FAN           0     // Channel 0: Waist Fans
#define PWM_CH_TEC           1     // Channel 1: Peltier TEC
#define PWM_CH_SHOULDER_FAN  2     // Channel 2: Shoulder Neck/Face Fans

#define MAX_HEATSINK_TEMP    55.0  // อุณหภูมิฮีตซิงก์สูงสุดที่ยอมรับได้ (°C)
#define MAX_BATT_TEMP        45.0  // อุณหภูมิแบตเตอรี่สูงสุดที่ยอมรับได้ (°C)
#define BATT_CUTOFF_VOLTS    9.6   // แรงดันตัดการทำงานเพื่อถนอมเซลล์ Li-ion (3.2V/cell)

// ================= GLOBAL OBJECTS & STRUCTS =================
Adafruit_BME280 bme; // BME280 บน I2C Address 0x76

struct SystemTelemetry {
  float ambientTemp;        // อุณหภูมิแวดล้อม (°C)
  float humidity;           // ความชื้นสัมพัทธ์ (% RH)
  float heatIndex;          // ดัชนีความร้อน NOAA (°C)
  float dewPoint;           // อุณหภูมิจุดน้ำค้าง (°C)
  float skinTemp;           // อุณหภูมิผิวหนัง (°C)
  float heatsinkTemp;       // อุณหภูมิแผงระบายความร้อน (°C)
  float batteryTemp;        // อุณหภูมิเซลล์แบตเตอรี่ NTC (°C)
  uint16_t heartRate;       // อัตราการเต้นของหัวใจ (BPM)
  uint16_t fanRPM;          // ความเร็วรอบพัดลมเอวจริง
  uint8_t fanDuty;          // สัญญาณ PWM พัดลมเอว (0-255)
  uint8_t shoulderFanDuty;  // สัญญาณ PWM พัดลมบ่าไหล่ (0-255)
  uint8_t tecDuty;          // สัญญาณ PWM Peltier (0-255)
  float batteryVolts;       // แรงดันแบตเตอรี่ (V)
  bool shoulderFanEnabled;  // สถานะเปิด/ปิดพัดลมบ่า
  bool batteryCoolingActive;// สถานะหล่อเย็นแบตเตอรี่ BTMS
  bool condensationAlert;   // ธงเตือนความเสี่ยงหยดน้ำ
  bool heatStrokeAlert;     // ธงเตือนภัยสภาวะลมแดด
  bool emergencyShutdown;   // ธงตัดการทำงานฉุกเฉิน
} telemetry;

volatile uint32_t fanPulseCounter = 0;
portMUX_TYPE mux = portMUX_INITIALIZER_UNLOCKED;

// Interrupt Service Routine วัดรอบพัดลม
void IRAM_ATTR onFanTachPulse() {
  portENTER_CRITICAL_ISR(&mux);
  fanPulseCounter++;
  portEXIT_CRITICAL_ISR(&mux);
}

// ================= MATHEMATICAL CALCULATIONS =================

/**
 * คำนวณจุดน้ำค้าง (Dew Point) ตามสูตร Magnus-Tetens
 * ใช้เพื่อควบคุมแผ่น Peltier ไม่ให้อุณหภูมิต่ำกว่าจุดน้ำค้าง ป้องกันการเกิดเหงื่อหยดน้ำ
 */
float calculateDewPoint(float temp, float humidity) {
  float a = 17.625;
  float b = 243.04;
  float alpha = ((a * temp) / (b + temp)) + log(humidity / 100.0);
  return (b * alpha) / (a - alpha);
}

/**
 * คำนวณดัชนีความร้อน NOAA Steadman Heat Index
 */
float calculateHeatIndex(float tempC, float humidity) {
  float T = (tempC * 9.0 / 5.0) + 32.0; // แปลงเป็น Fahrenheit
  float R = humidity;
  float hi = 0.5 * (T + 61.0 + ((T - 68.0) * 1.2) + (R * 0.094));
  
  if (hi >= 80.0) {
    hi = -42.379 + 2.04901523 * T + 10.14333127 * R - 0.22475541 * T * R
         - 0.00683783 * T * T - 0.05481717 * R * R + 0.00122874 * T * T * R
         + 0.00085282 * T * R * R - 0.00000199 * T * T * R * R;
  }
  return (hi - 32.0) * 5.0 / 9.0; // แปลงกลับเป็น Celsius
}

// ================= FREERTOS TASKS =================

/**
 * Task 1 (Core 1): ตรวจวัดสภาพแวดล้อมและประมวลผลการปรับอากาศ 2 โซน
 */
void TaskCoolingControl(void *pvParameters) {
  TickType_t xLastWakeTime = xTaskGetTickCount();
  const TickType_t xFrequency = pdMS_TO_TICKS(100); // 10Hz

  for (;;) {
    // 1. อ่านข้อมูลเซนเซอร์สภาพแวดล้อม
    telemetry.ambientTemp = bme.readTemperature();
    telemetry.humidity = bme.readHumidity();
    
    // 2. คำนวณ Heat Index และ Dew Point
    telemetry.heatIndex = calculateHeatIndex(telemetry.ambientTemp, telemetry.humidity);
    telemetry.dewPoint = calculateDewPoint(telemetry.ambientTemp, telemetry.humidity);

    // 3. ปรับระดับความเย็นอัตโนมัติ (AI-Adaptive Auto Mode)
    if (!telemetry.emergencyShutdown) {
      if (telemetry.heatIndex < 28.0) {
        telemetry.fanDuty = 65;           // 25% ลมลำตัวเบา
        telemetry.shoulderFanDuty = 75;   // 30% ลมเป่าหน้าสบาย
        telemetry.tecDuty = 0;            // ปิด Peltier
      } else if (telemetry.heatIndex < 35.0) {
        telemetry.fanDuty = 115;          // 45% ลมปานกลาง
        telemetry.shoulderFanDuty = 140;  // 55% ลมเป่าหน้าคลายร้อน
        telemetry.tecDuty = 75;           // 30% ความเย็นกระดูกสันหลัง
      } else if (telemetry.heatIndex < 42.0) {
        telemetry.fanDuty = 180;          // 70% ลมแรง
        telemetry.shoulderFanDuty = 205;  // 80% ลมเป่าหน้าเข้มข้น
        telemetry.tecDuty = 160;          // 65% ความเย็นเข้มข้น
      } else {
        // ดัชนีความร้อนระดับอันตราย (>42°C)
        telemetry.fanDuty = 245;          // 96% ลมแรงสูงสุด
        telemetry.shoulderFanDuty = 255;  // 100% ลมเป่าหน้าเต็มกำลัง
        telemetry.tecDuty = 220;          // 86% ความเย็นเร่งด่วน
      }

      // ตรวจสอบสถานะการปิดพัดลมบ่าแบบแมนนวล
      if (!telemetry.shoulderFanEnabled) {
        telemetry.shoulderFanDuty = 0;
      }

      // 4. Anti-Condensation Watchdog Interlock
      float estimatedColdPlate = telemetry.ambientTemp - (telemetry.tecDuty / 255.0) * 14.0;
      if (estimatedColdPlate < (telemetry.dewPoint + 1.5)) {
        telemetry.condensationAlert = true;
        telemetry.tecDuty = max(0, (int)(((telemetry.ambientTemp - (telemetry.dewPoint + 1.5)) / 14.0) * 255.0));
      } else {
        telemetry.condensationAlert = false;
      }
    } else {
      // สภาวะฉุกเฉิน
      telemetry.fanDuty = 255;
      telemetry.shoulderFanDuty = 0;
      telemetry.tecDuty = 0;
    }

    // ขับสัญญาณ PWM ไปยังมอเตอร์พัดลมและแผ่นเย็น
    ledcWrite(PWM_CH_FAN, telemetry.fanDuty);
    ledcWrite(PWM_CH_SHOULDER_FAN, telemetry.shoulderFanDuty);
    ledcWrite(PWM_CH_TEC, telemetry.tecDuty);

    vTaskDelayUntil(&xLastWakeTime, xFrequency);
  }
}

/**
 * Task 2 (Core 0): ระบบเฝ้าระวังความปลอดภัยและสัญญาณชีพ (Safety Watchdog & BTMS)
 */
void TaskSafetyWatchdog(void *pvParameters) {
  TickType_t xLastWakeTime = xTaskGetTickCount();
  const TickType_t xFrequency = pdMS_TO_TICKS(50); // 20Hz

  for (;;) {
    // 1. อ่านแรงดันแบตเตอรี่ (ADC ผ่าน Voltage Divider)
    int rawBatt = analogRead(PIN_BATT_SENSE);
    telemetry.batteryVolts = (rawBatt / 4095.0) * 3.3 * ((30.0 + 10.0) / 10.0);

    // 2. อ่านอุณหภูมิเซลล์แบตเตอรี่ (BTMS NTC Thermistor บน GPIO 39)
    int rawBattNtc = analogRead(PIN_BATT_NTC);
    telemetry.batteryTemp = (rawBattNtc / 4095.0) * 65.0; // สเกลอุณหภูมิ NTC
    telemetry.batteryCoolingActive = telemetry.batteryTemp > 35.0;

    // 3. อ่านอุณหภูมิฮีตซิงก์แผ่น Peltier
    int rawNtc = analogRead(PIN_HEATSINK_NTC);
    telemetry.heatsinkTemp = (rawNtc / 4095.0) * 80.0;

    // ตรวจสอบความปลอดภัยระดับวิกฤต:
    if (telemetry.heatsinkTemp > MAX_HEATSINK_TEMP || 
        telemetry.batteryTemp > MAX_BATT_TEMP || 
        telemetry.batteryVolts < BATT_CUTOFF_VOLTS) {
      telemetry.emergencyShutdown = true;
      digitalWrite(PIN_BUZZER, HIGH);
    } else {
      telemetry.emergencyShutdown = false;
      digitalWrite(PIN_BUZZER, LOW);
    }

    // ตรวจสอบสภาวะเสี่ยงฮีทสโตรก
    if (telemetry.heatIndex > 44.0 && telemetry.heartRate > 135) {
      telemetry.heatStrokeAlert = true;
      digitalWrite(PIN_HAPTIC, (millis() % 500 < 250) ? HIGH : LOW);
    } else {
      telemetry.heatStrokeAlert = false;
      digitalWrite(PIN_HAPTIC, LOW);
    }

    // คำนวณความเร็วรอบพัดลมจริง (RPM) ทุกๆ 1 วินาที
    static uint32_t lastTachCheck = 0;
    if (millis() - lastTachCheck >= 1000) {
      portENTER_CRITICAL(&mux);
      uint32_t pulses = fanPulseCounter;
      fanPulseCounter = 0;
      portEXIT_CRITICAL(&mux);
      telemetry.fanRPM = (pulses / 2) * 60;
      lastTachCheck = millis();
    }

    vTaskDelayUntil(&xLastWakeTime, xFrequency);
  }
}

// ================= SETUP & MAIN =================
void setup() {
  Serial.begin(115200);
  Wire.begin(PIN_I2C_SDA, PIN_I2C_SCL);

  telemetry.shoulderFanEnabled = true;
  telemetry.batteryCoolingActive = true;

  pinMode(PIN_BUZZER, OUTPUT);
  pinMode(PIN_HAPTIC, OUTPUT);
  pinMode(PIN_FAN_TACH, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(PIN_FAN_TACH), onFanTachPulse, FALLING);

  // ตั้งค่าฮาร์ดแวร์ PWM (LEDC) สำหรับพัดลมเอว, พัดลมบ่า และ Peltier
  ledcSetup(PWM_CH_FAN, PWM_FREQ_FAN, PWM_RES);
  ledcAttachPin(PIN_FAN_PWM, PWM_CH_FAN);

  ledcSetup(PWM_CH_SHOULDER_FAN, PWM_FREQ_FAN, PWM_RES);
  ledcAttachPin(PIN_SHOULDER_PWM, PWM_CH_SHOULDER_FAN);

  ledcSetup(PWM_CH_TEC, PWM_FREQ_TEC, PWM_RES);
  ledcAttachPin(PIN_TEC_PWM, PWM_CH_TEC);

  if (!bme.begin(0x76, &Wire)) {
    Serial.println(F("[ERR] BME280 not found! Check wiring."));
  }

  // สร้าง FreeRTOS Tasks กระจายลงสองคอร์ประมวลผล
  xTaskCreatePinnedToCore(TaskCoolingControl, "CoolingCtrl", 4096, NULL, 2, NULL, 1); // Core 1
  xTaskCreatePinnedToCore(TaskSafetyWatchdog, "SafetyWatch", 3072, NULL, 3, NULL, 0); // Core 0

  Serial.println(F("[OK] Smart Cooling Vest Dual-Core System Initialized."));
}

void loop() {
  vTaskDelay(pdMS_TO_TICKS(1000));
}

1. พัดลมบ่าไหล่เป่าหน้า/คอแบบแฮนด์ฟรี (Hands-Free Face Cooling):

ติดตั้งพัดลมไมโครเทอร์โบคู่ที่บ่าไหล่ ทำมุม 45° เล็งตรงสู่ใบหน้าและต้นคอ ช่วยระบายความร้อนบริเวณศีรษะและทางเดินหายใจทันที ผู้ใช้งานไม่ต้องพกหรือถือพัดลมมือถืออีกต่อไป ครบจบในเสื้อตัวเดียว

2. ระบบหล่อเย็นแบตเตอรี่เฉพาะทาง (Battery BTMS):

แบตเตอรี่ลิเธียมเมื่อจ่ายกระแสหนักกลางแดดจัดมีโอกาสร้อนสะสมสูง ระบบติดตั้งครีบอลูมิเนียม พัดลมระบายความร้อน และเซนเซอร์ NTC บน GPIO 39 คอยควบคุมอุณหภูมิไม่ให้เกิน 45°C เพื่อความปลอดภัยสูงสุดและยืดอายุการใช้งาน

3. สรีรวิทยาการระบายความร้อนสามโซน (3-Zone Thermal Comfort):

ผสานแรงดันลมพัดผ่านลำตัว เข้ากับแผ่นเย็นสัมผัสกระดูกสันหลัง (Peltier) และลมเป่าหน้า ทำให้ร่างกายรู้สึกเย็นสดชื่นทันใจ พร้อมอัลกอริทึม Dew-Point Watchdog ป้องกันหยดน้ำกลั่นตัว 100%

4. แถบสะท้อนแสง 3M Scotchlite & ความปลอดภัยกะกลางคืน:

เพิ่มทัศนวิสัยของผู้ปฏิบัติงานในที่มืด มองเห็นได้ไกลกว่า 300 เมตร พร้อมไฟสถานะฉุกเฉินและระบบตรวจจับการล้ม (Fall Detection)
